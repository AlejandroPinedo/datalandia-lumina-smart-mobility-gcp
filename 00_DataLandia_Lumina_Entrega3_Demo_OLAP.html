P3 consulta exclusivamente la tabla que ya está disponible en el proyecto: bigdata-herles-pinedo.lumina_gold_governed.fact_movilidad_multimodal_gold. Evidencia vigente

- evidencias/datos_existentes_latest.json: fuente, esquema, fecha, SQL, job ID y resultado. - evidencias/esquema_fuente_existente.json: metadatos consultados antes de agregar. - verificar_evidencia.py: contrasta el SQL y los totales de esa evidencia. - evidencias/archivo_pruebas_sinteticas/: pruebas anteriores retiradas de la entrega vigente.

La antigua prueba de 600 viajes no se utiliza como fuente, como respaldo de la API ni como evidencia de procesamiento del histórico del proyecto. Se conserva archivada para no borrar el rastro de lo que se había ejecutado.

Consultar de nuevo sin iniciar sesión otra vez

Con las credenciales ya guardadas, ejecutar en PowerShell:

.\P3\usar_datos_existentes.ps1

La consulta lee todos los registros de la tabla elegida, sin LIMIT ni tamaño de muestra fijo. Registra el esquema, agrega únicamente columnas disponibles y verifica la respuesta HTTP. Tiene un máximo de 5 GiB facturables por consulta. No ejecuta ingesta ni recalcula las capas Bronze/Silver/Gold del proyecto; utiliza el Gold preexistente como fuente del informe.

La sesión de Codex puede leer la exportación aunque su conexión de red esté restringida. Las credenciales permanecen en .runtime y están excluidas de Git.

Mostrar el tablero con la consulta ya exportada

python P3/demo_e2e.py --serve

Abrir http://127.0.0.1:8080/p3 y http://127.0.0.1:8080/api/v1/olap-kpis. El modo project_export sirve el resultado identificado de la consulta guardada, con tabla, job ID y fecha de consulta. No se presenta como una consulta remota en vivo. Si falta esa evidencia o corresponde a datos sintéticos, la API responde unavailable; el tablero muestra datos no disponibles, sin cifras de respaldo.

existing_project_data.py
"""Consultas de solo lectura sobre tablas existentes; sin generadores ni valores suplentes."""
from datetime import datetime, timezone
import hashlib
import json
import math
import re
import time


def validate_table_id(value):
    if not re.fullmatch(r'[a-z][a-z0-9-]*\.[A-Za-z_][A-Za-z0-9_]*\.[A-Za-z_][A-Za-z0-9_]*',value):
        raise ValueError('Se requiere project.dataset.table sin SQL adicional')
    return value


def build_query(table_id, columns):
    validate_table_id(table_id)
    names=set(columns)
    def pick(*candidates, required=True):
        found=next((n for n in candidates if n in names),None)
        if not found and required:
            raise ValueError('La fuente no contiene una columna requerida: '+', '.join(candidates))
        return f'`{found}`' if found else None
    mode=pick('transport_mode','modo_transporte',required=False)
    distance=pick('trip_distance_km','distancia_km')
    amount=pick('lumina_total_fare_usd','total_fare_usd','costo_total_usd','total_amount')
    duration=pick('lumina_duration_min','duration_minutes','duracion_min',required=False)
    passengers=pick('passenger_count','pasajeros',required=False)
    traffic=pick('traffic_match_found',required=False)
    iot=pick('iot_match_found',required=False)
    # Una etiqueta de ausencia de desglose no se presenta como un modo observado.
    mode_expr=mode or "'SIN_DESGLOSE_EN_FUENTE'"
    avg_duration=f'AVG({duration})' if duration else 'CAST(NULL AS FLOAT64)'
    avg_passengers=f'AVG({passengers})' if passengers else 'CAST(NULL AS FLOAT64)'
    traffic_count=f'COUNTIF({traffic})' if traffic else 'CAST(NULL AS INT64)'
    iot_count=f'COUNTIF({iot})' if iot else 'CAST(NULL AS INT64)'
    trip_id=pick('trip_id',required=False)
    unique=f'COUNT(DISTINCT {trip_id})' if trip_id else 'CAST(NULL AS INT64)'
    sql=f'''SELECT {mode_expr} AS transport_mode,COUNT(*) AS total_trips,
      {unique} AS unique_trip_ids,
      SUM({distance}) AS total_km,SUM({amount}) AS total_revenue_usd,
      SAFE_DIVIDE(SUM({amount}),SUM({distance})) AS revenue_per_km_usd,
      {avg_duration} AS avg_duration_min,{avg_passengers} AS avg_passengers,
      {traffic_count} AS traffic_matched_trips,{iot_count} AS iot_matched_trips,
      COUNTIF({distance} IS NULL) AS missing_distance_rows,
      COUNTIF({amount} IS NULL) AS missing_amount_rows,
      COUNTIF({distance} < 0) AS negative_distance_rows
    FROM `{table_id}` GROUP BY transport_mode ORDER BY transport_mode'''
    mapping={'mode':mode,'distance_km':distance,'amount_usd':amount,'duration_min':duration,
             'passengers':passengers,'traffic_match':traffic,'iot_match':iot,'trip_id':trip_id}
    return sql,mapping


def summarize(rows):
    # SUM conserva ausencia total; no convierte mediciones ausentes en ceros.
    def total(key, selected):
        values=[float(row[key]) for row in selected if row.get(key) is not None]
        if not values: return None
        if any(not math.isfinite(value) for value in values):
            raise ValueError('La fuente contiene totales no finitos')
        return round(sum(values),2)
    lumina=[row for row in rows if row.get('transport_mode')!='PERSONAL_CAR_BENCHMARK']
    return {'total_trips':sum(int(row['total_trips']) for row in rows),
            'total_km':total('total_km',rows),
            'total_revenue_usd':total('total_revenue_usd',lumina)}


def read_existing(client, table_id, job_config=None):
    start=time.monotonic()
    table_id=validate_table_id(table_id)
    table=client.get_table(table_id,timeout=20)
    columns=[field.name for field in table.schema]
    sql,mapping=build_query(table_id,columns)
    job=client.query(sql,job_config=job_config,location=table.location)
    rows=json.loads(json.dumps([dict(row) for row in job.result(timeout=120)],default=str))
    if not rows: raise ValueError('La tabla existente no devolvió filas; no se generan datos de respaldo')
    now=datetime.now(timezone.utc).isoformat()
    meta={'table_id':table_id,'table_type':table.table_type,'num_rows_at_inspection':table.num_rows,
          'num_bytes_at_inspection':table.num_bytes,'location':table.location,
          'modified':table.modified.isoformat() if table.modified else None,
          'description':table.description,'schema':[{'name':f.name,'type':f.field_type} for f in table.schema]}
    return {'status':'ok','backend':'bigquery_existing','live_query':True,
            'source':'bigquery:'+table_id,'data_origin':'existing_project_data',
            'provenance_status':'not_independently_verified',
            'run_id':'existing_'+datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S'),
            'processed_at':meta['modified'],'queried_at':now,
            'summary':summarize(rows),'by_mode':rows,'source_table':meta,
            'column_mapping':mapping,'job_id':job.job_id,
            'sql_sha256':hashlib.sha256(sql.encode()).hexdigest(),
            'query_sql':sql,'latency_ms':round((time.monotonic()-start)*1000,2),
            'source_modified_by_this_process':False,
            'limitation':'Se usan los registros ya existentes; no se certifica su procedencia externa por el solo hecho de estar en BigQuery.'}

from datetime import datetime,timezone
import json
import logging
import os
from pathlib import Path
import time
from existing_project_data import read_existing

ROOT=Path(__file__).resolve().parents[1]


def load_olap_report():
    started=time.monotonic()
    backend=os.getenv('LUMINA_DATA_BACKEND','project_export')
    try:
        if backend=='project_export':
            path=Path(os.getenv('LUMINA_PROJECT_REPORT',str(ROOT/'P3/evidencias/datos_existentes_latest.json')))
            data=json.loads(path.read_text(encoding='utf-8'))
            if data.get('data_origin')!='existing_project_data' or data.get('status')!='ok':
                raise ValueError('La API solo admite una consulta verificada a las tablas existentes')
            if not data.get('source','').startswith('bigquery:') or not data.get('job_id'):
                raise ValueError('Faltan la tabla fuente o el job de consulta')
            data['backend']='project_export'
            data['live_query']=False
            data['served_at']=datetime.now(timezone.utc).isoformat()
            data['latency_ms']=round((time.monotonic()-started)*1000,2)
            return data
        if backend in ('bigquery_existing','bigquery'):
            from google.cloud import bigquery
            import config
            source=os.getenv('LUMINA_SOURCE_TABLE',
                f'{config.GCP_PROJECT_ID}.{config.BQ_DATASET_GOLD}.{config.BQ_GOLD_FACT_TRIPS}')
            with bigquery.Client(project=config.GCP_PROJECT_ID) as client:
                return read_existing(client,source,
                    bigquery.QueryJobConfig(maximum_bytes_billed=5*1024**3))
        raise ValueError('P3 no admite generadores, bases de demo ni datos suplentes')
    except Exception as exc:
        logging.getLogger('lumina.olap').warning('Datos existentes no disponibles (%s)',type(exc).__name__)
        return {'status':'unavailable','backend':backend,'data_origin':None,
                'reason':'No hay una consulta a las tablas existentes disponible. Ejecuta P3/usar_datos_existentes.ps1.',
                'by_mode':[],'summary':None}

usar_datos_existentes.py
"""Lee el Gold existente y verifica su exposición por API. No genera ni carga viajes."""
import argparse
from datetime import datetime,timezone
import importlib
import json
import os
from pathlib import Path
import re
import sys
import threading
import urllib.request

ROOT=Path(__file__).resolve().parents[1]
CODE=ROOT/'Avance1_P1_Lumina_Ingesta_Storage'
sys.path.insert(0,str(CODE))
from existing_project_data import read_existing,validate_table_id
import config


def configure_credentials():
    packages=ROOT/'.runtime/python-gcp'
    if packages.is_dir(): sys.path.insert(0,str(packages))
    credentials=ROOT/'.runtime/gcloud/application_default_credentials.json'
    if credentials.is_file(): os.environ.setdefault('GOOGLE_APPLICATION_CREDENTIALS',str(credentials))
    os.environ.setdefault('CLOUDSDK_CONFIG',str(ROOT/'.runtime/gcloud'))


def verify_http(expected):
    api=importlib.import_module('08_app_exposicion_frontend_lumina')
    class Handler(api.StdlibLuminaHandler):
        def log_message(self,*args): pass
    server=api.HTTPServer(('127.0.0.1',0),Handler)
    worker=threading.Thread(target=server.serve_forever,daemon=True);worker.start()
    prefix=f'http://127.0.0.1:{server.server_address[1]}'
    try:
        with urllib.request.urlopen(prefix+'/api/v1/olap-kpis',timeout=10) as response:
            report=json.load(response)
            if report.get('status')!='ok' or report.get('summary')!=expected['summary']:
                raise AssertionError('La API no coincide con la consulta de las tablas existentes')
            if report.get('job_id')!=expected['job_id'] or report.get('data_origin')!='existing_project_data':
                raise AssertionError('La API no identifica la fuente existente consultada')
        for route in ('/p3','/'):
            with urllib.request.urlopen(prefix+route,timeout=5) as response:
                text=response.read().decode('utf-8')
                if '/api/v1/olap-kpis' not in text or re.search(r'\{[{%]',text):
                    raise AssertionError('El consumidor no está conectado o tiene una plantilla sin resolver')
        return {'api_matches_existing_query':'passed','consumer_references_api':'passed'}
    finally:
        server.shutdown();server.server_close();worker.join(timeout=3)


def export_existing(source):
    configure_credentials()
    from google.cloud import bigquery
    source=validate_table_id(source)
    project=source.split('.')[0]
    evidence_dir=ROOT/'P3/evidencias';evidence_dir.mkdir(exist_ok=True)
    source_info=None
    try:
        with bigquery.Client(project=project) as client:
            table=client.get_table(source,timeout=20)
            source_info={'table_id':source,'num_rows_at_inspection':table.num_rows,
                         'description':table.description,
                         'schema':[{'name':field.name,'type':field.field_type} for field in table.schema]}
            (evidence_dir/'esquema_fuente_existente.json').write_text(
                json.dumps(source_info,ensure_ascii=False,indent=2,default=str),encoding='utf-8')
            report=read_existing(client,source,bigquery.QueryJobConfig(maximum_bytes_billed=5*1024**3))
    except Exception as exc:
        diagnostic={'status':'not_completed','source':source,'error_type':type(exc).__name__,
                    'message':str(exc),'source_info':source_info,
                    'checked_at':datetime.now(timezone.utc).isoformat()}
        # Diagnóstico local: no se publica en el repositorio ni incluye archivos de credenciales.
        runtime=ROOT/'.runtime';runtime.mkdir(exist_ok=True)
        (runtime/'consulta_existente_error.json').write_text(
            json.dumps(diagnostic,ensure_ascii=False,indent=2,default=str),encoding='utf-8')
        raise
    output=evidence_dir/'datos_existentes_latest.json'
    output.write_text(json.dumps(report,ensure_ascii=False,indent=2,default=str),encoding='utf-8')
    os.environ['LUMINA_DATA_BACKEND']='project_export'
    os.environ['LUMINA_PROJECT_REPORT']=str(output)
    report['verification']=verify_http(json.loads(output.read_text(encoding='utf-8')))
    output.write_text(json.dumps(report,ensure_ascii=False,indent=2,default=str),encoding='utf-8')
    return report


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--source-table',default=os.getenv('LUMINA_SOURCE_TABLE',
        f'{config.GCP_PROJECT_ID}.{config.BQ_DATASET_GOLD}.{config.BQ_GOLD_FACT_TRIPS}'))
    parser.add_argument('--serve',action='store_true')
    args=parser.parse_args()
    report=export_existing(args.source_table)
    print(json.dumps({'status':report['status'],'source':report['source'],'summary':report['summary'],
                      'job_id':report['job_id'],'verification':report['verification'],
                      'source_modified_by_this_process':False},ensure_ascii=False,indent=2))
    if args.serve:
        api=importlib.import_module('08_app_exposicion_frontend_lumina')
        print(f'Abrir http://127.0.0.1:{config.FLASK_PORT}/p3')
        api.HTTPServer((config.FLASK_HOST,config.FLASK_PORT),api.StdlibLuminaHandler).serve_forever()


if __name__=='__main__': main()
verificar_evidencia.py
"""Verifica únicamente la evidencia de lectura de tablas existentes."""
import hashlib
import json
from pathlib import Path
import sys

ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'Avance1_P1_Lumina_Ingesta_Storage'))
from existing_project_data import build_query,summarize


def verify():
    path=ROOT/'P3/evidencias/datos_existentes_latest.json'
    data=json.loads(path.read_text(encoding='utf-8'))
    if data.get('status')!='ok' or data.get('data_origin')!='existing_project_data':
        raise ValueError('No hay evidencia válida de las tablas existentes')
    if data.get('source_modified_by_this_process') is not False:
        raise ValueError('Falta verificar que no se modificó la fuente')
    table=data['source_table']
    sql,_=build_query(table['table_id'],[field['name'] for field in table['schema']])
    if hashlib.sha256(sql.encode()).hexdigest()!=data['sql_sha256']:
        raise ValueError('El SQL no coincide con la evidencia')
    if summarize(data['by_mode'])!=data['summary']:
        raise ValueError('Los totales no coinciden con las filas consultadas')
    verification=data.get('verification',{})
    if not data.get('job_id') or any(verification.get(key)!='passed' for key in
            ('api_matches_existing_query','consumer_references_api')):
        raise ValueError('Falta la consulta o la verificación HTTP')
    metadata_rows=table.get('num_rows_at_inspection')
    if metadata_rows is not None and metadata_rows!=data['summary']['total_trips']:
        raise ValueError('El conteo consultado no coincide con los metadatos registrados')
    return {'status':'passed','source':data['source'],'source_rows':data['summary']['total_trips'],
            'job_id':data['job_id'],'generated_data_used':False,
            'provenance_status':data['provenance_status']}


if __name__=='__main__': print(json.dumps(verify(),ensure_ascii=False,indent=2))
calculadora_consumo.py
"""Estimación editable del primer año para el piloto OLAP, sin prometer costos de producción."""
import argparse
import json
import math
from pathlib import Path


def calculate(config):
    usage, prices = config['usage_monthly'], config['prices_usd']
    values = list(usage.values()) + list(prices.values()) + [config['months'],config['contingency_pct']]
    if any(not isinstance(v,(float,int)) or not math.isfinite(v) or v<0 for v in values):
        raise ValueError('Los consumos y precios deben ser números finitos no negativos')
    if config.get('free_tier_applied'):
        raise ValueError('Este escenario conservador no descuenta cuotas gratuitas compartidas')
    lines = {
        'bigquery_queries':usage['bigquery_scanned_tib']*prices['bigquery_per_tib'],
        'bigquery_storage':usage['bigquery_logical_storage_gib']*prices['bigquery_storage_per_gib_month'],
        'cloud_run_cpu':usage['cloud_run_vcpu_seconds']*prices['cloud_run_per_vcpu_second'],
        'cloud_run_memory':usage['cloud_run_gib_seconds']*prices['cloud_run_per_gib_second'],
        'cloud_run_requests':usage['cloud_run_requests']/1e6*prices['cloud_run_per_million_requests'],
    }
    subtotal=sum(lines.values())
    monthly=subtotal*(1+config['contingency_pct']/100)
    return {'scenario':config['scenario'],'currency':config['currency'],
            'monthly_lines':{k:round(v,6) for k,v in lines.items()},'monthly_subtotal':round(subtotal,4),
            'contingency_pct':config['contingency_pct'],'monthly_with_contingency':round(monthly,4),
            'first_year_with_contingency':round(monthly*config['months'],2),
            'sources':config['sources'],'excluded':config['excluded'],
            'limitation':'Estimación del piloto con supuestos; no es factura, presupuesto aprobado ni TCO metropolitano.'}


if __name__=='__main__':
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--inputs',type=Path,default=Path(__file__).with_name('costos.json'))
    args=parser.parse_args()
    result=calculate(json.loads(args.inputs.read_text(encoding='utf-8')))
    output=Path(__file__).parent/'evidencias'/'estimacion_anual.json'
    output.parent.mkdir(exist_ok=True)
    output.write_text(json.dumps(result,ensure_ascii=False,indent=2),encoding='utf-8')
    print(json.dumps(result,ensure_ascii=False,indent=2))
